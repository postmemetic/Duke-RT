cmake_minimum_required(VERSION 3.30)
project(DukeRTNriRuntime LANGUAGES C CXX)
if(NOT WIN32 OR NOT RAZE_SOURCE_ROOT)
    message(FATAL_ERROR "This runtime owner requires Windows and RAZE_SOURCE_ROOT.")
endif()
set(_vendor "${RAZE_SOURCE_ROOT}/libraries/NRIFramework/External/NRI")
if(NOT EXISTS "${_vendor}/Include/NRI.h")
    message(FATAL_ERROR "Vendored NRI source is missing: ${_vendor}")
endif()
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin/$<CONFIG>")
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin/$<CONFIG>")
foreach(_config Debug Release RelWithDebInfo MinSizeRel)
    string(TOUPPER "${_config}" _upper)
    set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_${_upper} "${CMAKE_BINARY_DIR}/bin/${_config}")
    set(CMAKE_LIBRARY_OUTPUT_DIRECTORY_${_upper} "${CMAKE_BINARY_DIR}/bin/${_config}")
endforeach()
set(NRI_SHADERS_PATH "${CMAKE_BINARY_DIR}/shaders/$<CONFIG>" CACHE STRING "" FORCE)
set(FETCHCONTENT_SOURCE_DIR_SHADERMAKE "${RAZE_SOURCE_ROOT}/libraries/ShaderMake" CACHE PATH "" FORCE)
set(NRI_STATIC_LIBRARY OFF CACHE BOOL "" FORCE)
set(NRI_ENABLE_IMGUI_EXTENSION OFF CACHE BOOL "" FORCE)
set(NRI_AGILITY_SDK_VERSION_MAJOR "619" CACHE STRING "" FORCE)
set(NRI_AGILITY_SDK_VERSION_MINOR "0" CACHE STRING "" FORCE)
set(_arm FALSE)
if(CMAKE_GENERATOR_PLATFORM MATCHES "ARM64" OR CMAKE_SYSTEM_PROCESSOR MATCHES "aarch64|arm64|ARM64")
    set(_arm TRUE)
    set(_arch arm64)
elseif(CMAKE_SIZEOF_VOID_P EQUAL 8)
    set(_arch x64)
else()
    message(FATAL_ERROR "Duke-RT's Windows NRI runtime requires x64 or ARM64.")
endif()
set(_always_on D3D11_SUPPORT D3D12_SUPPORT VK_SUPPORT NONE_SUPPORT VALIDATION_SUPPORT
    DEBUG_NAMES_AND_ANNOTATIONS NVTX_SUPPORT AGILITY_SDK_SUPPORT NIS_SDK)
foreach(_name IN LISTS _always_on)
    set(NRI_ENABLE_${_name} ON CACHE BOOL "" FORCE)
endforeach()
foreach(_name NGX_SDK FFX_SDK XESS_SDK AMDAGS NVAPI)
    if(_arm)
        set(NRI_ENABLE_${_name} OFF CACHE BOOL "" FORCE)
    else()
        set(NRI_ENABLE_${_name} ON CACHE BOOL "" FORCE)
    endif()
endforeach()
set(NRI_STREAMER_THREAD_SAFE ON CACHE BOOL "" FORCE)
if(RAZE_NRI_DEPENDENCY_SOURCE_DIR)
    if(NOT IS_DIRECTORY "${RAZE_NRI_DEPENDENCY_SOURCE_DIR}")
        message(FATAL_ERROR "Dependency source cache is not a directory: ${RAZE_NRI_DEPENDENCY_SOURCE_DIR}")
    endif()
    foreach(_dep agilitysdk amdags d3d12ma dxc ffx ngx nvapi nvtx vma vulkan_headers xess)
        if(IS_DIRECTORY "${RAZE_NRI_DEPENDENCY_SOURCE_DIR}/${_dep}-src")
            string(TOUPPER "${_dep}" _upper)
            set(FETCHCONTENT_SOURCE_DIR_${_upper} "${RAZE_NRI_DEPENDENCY_SOURCE_DIR}/${_dep}-src" CACHE PATH "" FORCE)
        endif()
    endforeach()
endif()
add_subdirectory("${_vendor}" vendor)

# Capture dependency source paths while they are in scope. Stage directly from
# these sources so deleting a copied companion never requires relinking NRI.
foreach(_dep agilitysdk amdags ffx ngx xess)
    get_directory_property(${_dep}_SOURCE_DIR DIRECTORY "${_vendor}" DEFINITION ${_dep}_SOURCE_DIR)
endforeach()
set(_inputs "set(NRI_INPUT_CONFIGURATION [==[$<CONFIG>]==])\n")
macro(_input name value)
    string(APPEND _inputs "set(${name} [==[${value}]==])\n")
endmacro()
_input(NRI_INPUT_ARCHITECTURE "${_arch}")
_input(NRI_INPUT_COMPILER_ID "${CMAKE_CXX_COMPILER_ID}")
_input(NRI_INPUT_COMPILER_VERSION "${CMAKE_CXX_COMPILER_VERSION}")
_input(NRI_INPUT_COMPILER_PATH "${CMAKE_CXX_COMPILER}")
_input(NRI_INPUT_SOURCE_ROOT "${RAZE_SOURCE_ROOT}")
_input(NRI_INPUT_AGILITY_VERSION "619.0")
set(_options)
foreach(_name IN LISTS _always_on)
    list(APPEND _options NRI_ENABLE_${_name})
endforeach()
list(APPEND _options NRI_ENABLE_NGX_SDK NRI_ENABLE_FFX_SDK NRI_ENABLE_XESS_SDK
    NRI_ENABLE_AMDAGS NRI_ENABLE_NVAPI NRI_STREAMER_THREAD_SAFE NRI_STATIC_LIBRARY NRI_ENABLE_IMGUI_EXTENSION)
_input(NRI_INPUT_OPTIONS "${_options}")
foreach(_option IN LISTS _options)
    _input(${_option} "${${_option}}")
endforeach()
set(_paths)
macro(_runtime path source)
    list(APPEND _paths "${path}")
    string(MAKE_C_IDENTIFIER "${path}" _key)
    _input(NRI_INPUT_FILE_${_key} "${source}")
endmacro()
_runtime("NRI.dll" "$<TARGET_FILE:NRI>")
_runtime("NRI-LICENSE.txt" "${_vendor}/LICENSE.txt")
_runtime("AgilitySDK/D3D12Core.dll" "${agilitysdk_SOURCE_DIR}/build/native/bin/${_arch}/D3D12Core.dll")
_runtime("AgilitySDK/d3d12SDKLayers.dll" "${agilitysdk_SOURCE_DIR}/build/native/bin/${_arch}/d3d12SDKLayers.dll")
if(NOT _arm)
    _runtime("amd_ags_x64.dll" "${amdags_SOURCE_DIR}/ags_lib/lib/amd_ags_x64.dll")
    _runtime("amd_fidelityfx_dx12.dll" "${ffx_SOURCE_DIR}/PrebuiltSignedDLL/amd_fidelityfx_dx12.dll")
    _runtime("amd_fidelityfx_vk.dll" "${ffx_SOURCE_DIR}/PrebuiltSignedDLL/amd_fidelityfx_vk.dll")
    _runtime("FidelityFX-SDK-LICENSE.txt" "${ffx_SOURCE_DIR}/sdk/LICENSE.txt")
    _runtime("libxess.dll" "${xess_SOURCE_DIR}/bin/libxess.dll")
    _runtime("nvngx_dlss.dll" "${ngx_SOURCE_DIR}/lib/Windows_x86_64/$<IF:$<CONFIG:Debug>,dev,rel>/nvngx_dlss.dll")
    _runtime("nvngx_dlssd.dll" "${ngx_SOURCE_DIR}/lib/Windows_x86_64/$<IF:$<CONFIG:Debug>,dev,rel>/nvngx_dlssd.dll")
endif()
_input(NRI_INPUT_PATHS "${_paths}")

# Embed the exact source/build identity into the compiled DLL. A fresh staging
# invocation cannot give an old DLL a receipt for newly edited source.
include("${RAZE_SOURCE_ROOT}/cmake/NriRuntimeManifest.cmake")
nri_source_fingerprint("${RAZE_SOURCE_ROOT}" _source_hash _nri_version)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${_source_hash_FILES})
nri_build_context("${_arch}" "${CMAKE_CXX_COMPILER_ID}" "${CMAKE_CXX_COMPILER_VERSION}"
    "${CMAKE_CXX_COMPILER}" "619.0" _context_hash)
set(_marker "DukeRTNriBuildIdentity:${_source_hash}:$<CONFIG>:${_context_hash}")
file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/nri-runtime-identity-$<CONFIG>.cpp"
    CONTENT "extern \"C\" __declspec(dllexport) const char DukeRTNriBuildIdentity[] = \"${_marker}\";\n")
target_sources(NRI PRIVATE "${CMAKE_BINARY_DIR}/nri-runtime-identity-$<CONFIG>.cpp")
_input(NRI_INPUT_SOURCE_SHA256 "${_source_hash}")
_input(NRI_INPUT_CONTEXT_SHA256 "${_context_hash}")
file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/nri-build-inputs-$<CONFIG>.cmake" CONTENT "${_inputs}")
